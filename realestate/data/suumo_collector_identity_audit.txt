    @property
    def listing_id(self) -> str:
        raw = f"{self.source}|{canonical_url(self.url)}"
        return hashlib.sha1(raw.encode("utf-8")).hexdigest()[:20]

    @property
    def property_id(self) -> str:
        parts = [
            normalize_for_key(self.address or ""),
            round_key(self.land_area_sqm),
            round_key(self.building_area_sqm),
            normalize_for_key(self.built_year_month or ""),
            normalize_for_key(self.layout or ""),
        ]
        meaningful = [p for p in parts if p]
        if len(meaningful) < 2:
            return f"listing:{self.listing_id}"
        return hashlib.sha1("|".join(parts).encode("utf-8")).hexdigest()[:20]

    def as_state(self) -> dict[str, Any]:
        return {
            "listing_id": self.listing_id,
            "property_id": self.property_id,
            "source": self.source,
            "source_listing_id": self.source_listing_id,
            "url": canonical_url(self.url),
            "ward": self.ward,
            "title": self.title,

def canonical_url(url: str) -> str:
    parsed = urlparse(url)
    return urlunparse((parsed.scheme or "https", parsed.netloc.lower(), parsed.path, "", "", ""))


def navigation_url(url: str) -> str:
    """Normalize a crawl URL while preserving pagination query parameters."""
    parsed = urlparse(url)
    return urlunparse((parsed.scheme or "https", parsed.netloc.lower(), parsed.path, "", parsed.query, ""))

def collect_fudousan_import() -> tuple[list[Listing], dict[str, Any]]:
    path = IMPORTS_DIR / "fudousan_japan.json"
    if not path.exists():
        return [], {
            "source": "fudousan_japan",
            "success": True,
            "listing_rows": 0,
            "price_rows": 0,
            "coverage_note": "No official/manual import file present; full coverage not guaranteed.",
        }
    try:
        payload = json.loads(path.read_text(encoding="utf-8"))
        rows = payload.get("listings", []) if isinstance(payload, dict) else payload
        listings: list[Listing] = []
        for row in rows:
            url = str(row.get("url") or "").strip()
            if not url:
                continue
            source_id = str(row.get("source_listing_id") or hashlib.sha1(url.encode()).hexdigest()[:12])
            listings.append(
                Listing(
                    source="fudousan_japan",
                    source_listing_id=source_id,
                    url=url,
                    ward=str(row.get("ward") or ""),
                    title=row.get("title"),
                    price_yen=row.get("price_yen") or parse_price_yen(row.get("price_text")),
                    price_text=row.get("price_text"),
                    address=row.get("address"),
                    access=row.get("access"),
                    land_area_sqm=row.get("land_area_sqm"),
                    building_area_sqm=row.get("building_area_sqm"),
                    layout=row.get("layout"),
                    built_year_month=row.get("built_year_month"),
                    broker=row.get("broker"),
                )
            )
        return listings, {
            "source": "fudousan_japan",
            "success": True,
            "listing_rows": len(listings),
            "price_rows": sum(item.price_yen is not None for item in listings),
            "coverage_note": "Imported from official/manual feed; full-site coverage not guaranteed.",
        }
    except Exception as exc:
        return [], {
            "source": "fudousan_japan",
            "success": False,
            "listing_rows": 0,
            "price_rows": 0,
            "errors": [f"{type(exc).__name__}: {exc}"],
        }


def load_json(path: Path, default: Any) -> Any:
    if not path.exists():
        return default
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except Exception:
        return default


def write_json(path: Path, value: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_suffix(path.suffix + ".tmp")
    temporary.write_text(json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True), encoding="utf-8")
    temporary.replace(path)


def load_jsonl(path: Path) -> list[dict[str, Any]]:
    if not path.exists():
        return []
    records = []
    for line in path.read_text(encoding="utf-8").splitlines():
        if not line.strip():
            continue
        try:
            records.append(json.loads(line))
        except json.JSONDecodeError:
            continue
    return records


def write_jsonl(path: Path, records: Iterable[dict[str, Any]]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    lines = [json.dumps(item, ensure_ascii=False, sort_keys=True) for item in records]
    content = "\n".join(lines) + ("\n" if lines else "")
    temporary = path.with_suffix(path.suffix + ".tmp")
    temporary.write_text(content, encoding="utf-8")
    temporary.replace(path)

def merge_month_observations(observations: list[dict[str, Any]]) -> None:
    path = HISTORY_DIR / f"{MONTH_KEY}.jsonl"
    existing = load_jsonl(path)
    by_key = {(row.get("date"), row.get("listing_id")): row for row in existing}
    for row in observations:
        by_key[(row["date"], row["listing_id"])] = row
    ordered = sorted(by_key.values(), key=lambda row: (row.get("date", ""), row.get("listing_id", "")))
    write_jsonl(path, ordered)


def append_run(run: dict[str, Any]) -> None:
    path = RUNS_DIR / f"{MONTH_KEY}.jsonl"
    rows = load_jsonl(path)
    rows.append(run)
    rows = sorted(rows, key=lambda row: row.get("started_at", ""))
    write_jsonl(path, rows)


def export_dashboard(state: dict[str, Any], latest_run: dict[str, Any]) -> None:
    history_by_listing: dict[str, list[list[Any]]] = defaultdict(list)
    for path in sorted(HISTORY_DIR.glob("*.jsonl")):
        for row in load_jsonl(path):
            listing_id = row.get("listing_id")
            if not listing_id or row.get("price_yen") is None:
                continue
            history_by_listing[listing_id].append([row.get("date"), row.get("price_yen")])

    listings: list[dict[str, Any]] = []
    for listing_id, row in state.get("listings", {}).items():
        item = dict(row)
        history = sorted(history_by_listing.get(listing_id, []), key=lambda point: point[0] or "")
        # Deduplicate any duplicate dates from older schemas.
        compact = {point[0]: point[1] for point in history if point[0]}
        item["price_history"] = [[day, compact[day]] for day in sorted(compact)]
        first_price = item["price_history"][0][1] if item["price_history"] else None
        current_price = item.get("price_yen")
        if first_price and current_price is not None:
            item["price_change_yen"] = current_price - first_price
            item["price_change_pct"] = round((current_price - first_price) / first_price * 100, 2)
        else:
            item["price_change_yen"] = None
            item["price_change_pct"] = None
        try:
            first_seen_date = dt.datetime.fromisoformat(item["first_seen"]).date()
            item["observed_days"] = (NOW.date() - first_seen_date).days + 1
        except Exception:
            item["observed_days"] = None
        listings.append(item)

    listings.sort(
        key=lambda item: (
            not bool(item.get("active")),
            item.get("ward") or "",
            item.get("price_yen") is None,
            item.get("price_yen") or 0,
        )
    )
    active = [item for item in listings if item.get("active")]
    summary = {
        "active_listings": len(active),
        "all_known_listings": len(listings),
        "active_by_source": dict(sorted(_count_by(active, "source").items())),
        "active_by_ward": dict(sorted(_count_by(active, "ward").items())),
        "priced_active_listings": sum(item.get("price_yen") is not None for item in active),
        "price_coverage_pct": round(
            100 * sum(item.get("price_yen") is not None for item in active) / len(active), 1
        )
        if active
        else 0.0,
        "latest_new_count": latest_run.get("metrics", {}).get("new_count", 0),
        "latest_price_changed_count": latest_run.get("metrics", {}).get("price_changed_count", 0),
        "latest_ended_count": latest_run.get("metrics", {}).get("ended_count", 0),
    }
    dashboard = {
        "schema_version": 2,
        "generated_at": NOW.isoformat(),
        "observed_date": OBSERVED_DATE,
        "summary": summary,
        "coverage": state.get("coverage", {}),
        "latest_run": latest_run,
        "listings": listings,
    }
    write_json(DASHBOARD_PATH, dashboard)
